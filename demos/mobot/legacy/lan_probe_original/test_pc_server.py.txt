"""
test_pc_server.py - 메인 PC 수신 테스트 서버
스마트 팩토리 통신규약 v0.1.0 로컬 통신 검증용 (DB/장비 의존성 없음)

실행: python test_pc_server.py
"""
import json
import sys
from datetime import datetime, timezone
from http.server import BaseHTTPRequestHandler, HTTPServer

HOST = "0.0.0.0"
PORT = 5000

# ── 콘솔 컬러 (Windows/Linux 공용) ─────────────────────────────────────
GRN  = "\033[92m"
YEL  = "\033[93m"
CYN  = "\033[96m"
RED  = "\033[91m"
BLD  = "\033[1m"
RST  = "\033[0m"

def now_kst():
    utc = datetime.now(timezone.utc)
    return utc.strftime("%H:%M:%S.%f")[:-3]

def ok(msg):   print(f"{GRN}[OK]{RST}  {msg}")
def info(msg): print(f"{CYN}[>>]{RST}  {msg}")
def warn(msg): print(f"{YEL}[??]{RST}  {msg}")


class Handler(BaseHTTPRequestHandler):

    def log_message(self, fmt, *args):
        pass  # 기본 로그 억제 (커스텀 출력 사용)

    def _send(self, status: int, body: dict):
        raw = json.dumps(body, ensure_ascii=False).encode()
        self.send_response(status)
        self.send_header("Content-Type", "application/json; charset=utf-8")
        self.send_header("Content-Length", str(len(raw)))
        self.send_header("X-Mobot-Contract", "0.1.0")
        self.end_headers()
        self.wfile.write(raw)

    def do_GET(self):
        client_ip = self.client_address[0]
        if self.path == "/ping":
            ok(f"[{now_kst()}] PING ← {BLD}{client_ip}{RST}")
            self._send(200, {"status": "ok", "server": "test_pc_server", "contract": "0.1.0"})
        else:
            warn(f"[{now_kst()}] GET {self.path} ← {client_ip} (미등록 경로)")
            self._send(404, {"error": "not found"})

    def do_POST(self):
        client_ip = self.client_address[0]
        length = int(self.headers.get("Content-Length", 0))
        raw = self.rfile.read(length) if length else b""

        if self.path == "/api/v1/device-events":
            try:
                payload = json.loads(raw)
            except Exception:
                warn(f"[{now_kst()}] POST /api/v1/device-events ← {client_ip} JSON 파싱 실패")
                self._send(400, {"code": "BAD_REQUEST", "message": "JSON 형식 오류"})
                return

            event_id  = payload.get("event_id",  "?")
            device_id = payload.get("device_id", "?")
            body      = payload.get("body", {})
            evt_type  = body.get("event_type", "?")
            message   = body.get("message", "")

            print()
            print(f"{BLD}{'─'*55}{RST}")
            ok(f"[{now_kst()}] 이벤트 수신 ← {BLD}{client_ip}{RST}")
            info(f"device_id  : {BLD}{device_id}{RST}")
            info(f"event_id   : {event_id}")
            info(f"event_type : {BLD}{YEL}{evt_type}{RST}")
            info(f"message    : {message}")
            print(f"{BLD}{'─'*55}{RST}")
            print()

            self._send(202, {
                "status": "accepted",
                "event_id": event_id,
                "received_at": datetime.now(timezone.utc).isoformat().replace("+00:00", "Z")
            })
        else:
            warn(f"[{now_kst()}] POST {self.path} ← {client_ip} (미등록 경로)")
            self._send(404, {"error": "not found"})


if __name__ == "__main__":
    # Windows 콘솔 ANSI 색상 활성화
    if sys.platform == "win32":
        import ctypes
        ctypes.windll.kernel32.SetConsoleMode(
            ctypes.windll.kernel32.GetStdHandle(-11), 7
        )

    server = HTTPServer((HOST, PORT), Handler)
    print()
    print(f"{BLD}{GRN}== 스마트팩토리 TCP/IP 테스트 서버 (통신규약 v0.1.0) =={RST}")
    print(f"   수신 주소 : {BLD}http://0.0.0.0:{PORT}{RST}  (내부망 전체 오픈)")
    print(f"   내 IP     : {BLD}192.168.0.175{RST}  (라파에서 이 주소로 접속)")
    print(f"   종료      : Ctrl+C")
    print()
    print(f"   지원 엔드포인트:")
    print(f"   {CYN}GET  /ping                {RST}← 헬스체크 (RTT 측정용)")
    print(f"   {CYN}POST /api/v1/device-events{RST}← 장비 이벤트 수신 (v0.1.0)")
    print()
    print(f"{YEL}>>> 라파에서 test_raspi_client.py 를 실행하세요 <<<{RST}")
    print()

    try:
        server.serve_forever()
    except KeyboardInterrupt:
        print(f"\n{YEL}[서버 종료]{RST}")
