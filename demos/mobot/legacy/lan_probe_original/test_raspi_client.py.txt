"""
test_raspi_client.py - 라즈베리파이 송신 테스트 클라이언트
스마트 팩토리 통신규약 v0.1.0 로컬 통신 검증용

의존성: Python 표준 라이브러리만 사용 (pip install 불필요)

실행 예시:
  python3 test_raspi_client.py
  python3 test_raspi_client.py --server http://192.168.0.175:5000
  python3 test_raspi_client.py --server http://192.168.0.175:5000 --device LINE
"""
import argparse
import json
import os
import socket
import sys
import time
import urllib.error
import urllib.request
from uuid import uuid4

# ── 콘솔 컬러 ─────────────────────────────────────────────────────────
GRN = "\033[92m"
YEL = "\033[93m"
CYN = "\033[96m"
RED = "\033[91m"
BLD = "\033[1m"
RST = "\033[0m"

# 환경변수 우선 적용, 없으면 기본값 (규약 공식 장비 ID: LINE / ARM / AGV)
DEFAULT_SERVER = os.environ.get("MOBOT_SERVER_URL", "http://192.168.0.175:5000")
DEFAULT_DEVICE_ID = os.environ.get("MOBOT_DEVICE_ID", "LINE")

# 부팅 시작 시 고정되는 boot_id (실제 raspi_agent.py 와 동일 방식)
BOOT_ID = str(uuid4())


def get_local_ip() -> str:
    """현재 기기(라즈베리파이)의 활성화된 네트워크 인터페이스 로컬 IP를 동적으로 조회"""
    try:
        s = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
        # 외부 통신 시도 형태로 소켓을 열어 라우팅 테이블 상의 로컬 IP 획득 (실제 패킷 미전송)
        s.connect(("8.8.8.8", 80))
        ip = s.getsockname()[0]
        s.close()
        return ip
    except Exception:
        try:
            return socket.gethostbyname(socket.gethostname())
        except Exception:
            return "127.0.0.1"


def print_header(title: str):
    print(f"\n{BLD}{CYN}{'━'*52}{RST}")
    print(f"{BLD}{CYN}  {title}{RST}")
    print(f"{BLD}{CYN}{'━'*52}{RST}")


def test_ping(base_url: str) -> bool:
    """Step 1: GET /ping  →  왕복 지연(RTT) 측정"""
    print_header("STEP 1 / 2  :  Ping 헬스체크")
    url = f"{base_url}/ping"
    print(f"  요청  : {CYN}GET {url}{RST}")

    try:
        req = urllib.request.Request(url, method="GET")
        t0 = time.perf_counter()
        with urllib.request.urlopen(req, timeout=5) as res:
            rtt_ms = (time.perf_counter() - t0) * 1000
            body = json.loads(res.read())
            status = res.status

        if status == 200:
            print(f"  결과  : {GRN}{BLD}✓ 성공{RST}  (HTTP {status})")
            print(f"  RTT   : {BLD}{rtt_ms:.1f} ms{RST}")
            print(f"  응답  : {body}")
            return True
        else:
            print(f"  결과  : {YEL}예상 외 응답 (HTTP {status}){RST}")
            return False

    except urllib.error.URLError as e:
        print(f"  결과  : {RED}{BLD}✗ 실패{RST}  — {e.reason}")
        print(f"\n  {YEL}[확인 사항]{RST}")
        print(f"  · PC에서 test_pc_server.py 가 실행 중인지 확인")
        print(f"  · Windows 방화벽에서 TCP 5000 포트가 허용되어 있는지 확인")
        print(f"    (관리자 PowerShell: netsh advfirewall firewall add rule")
        print(f"     name=\"MOBOT-TEST\" dir=in action=allow protocol=TCP localport=5000)")
        return False


def test_device_event(base_url: str, device_id: str) -> bool:
    """Step 2: POST /api/v1/device-events  →  v0.1.0 DTO 전송"""
    print_header("STEP 2 / 2  :  v0.1.0 장비 이벤트 전송")
    url = f"{base_url}/api/v1/device-events"

    event_id = str(uuid4())
    payload = {
        "event_id":  event_id,
        "device_id": device_id,
        "boot_id":   BOOT_ID,
        "event_seq": 1,
        "body": {
            "event_type": "TEST_PING",
            "message":    "Raspi connection ok"
        }
    }

    print(f"  요청  : {CYN}POST {url}{RST}")
    print(f"  device_id  : {BLD}{device_id}{RST}")
    print(f"  event_id   : {event_id}")
    print(f"  boot_id    : {BOOT_ID}")
    print(f"  event_seq  : 1")
    print(f"  event_type : {BLD}TEST_PING{RST}")

    raw = json.dumps(payload, ensure_ascii=False).encode()
    req = urllib.request.Request(
        url,
        data=raw,
        method="POST",
        headers={
            "Content-Type":   "application/json; charset=utf-8",
            "Content-Length": str(len(raw)),
        }
    )

    try:
        t0 = time.perf_counter()
        with urllib.request.urlopen(req, timeout=5) as res:
            rtt_ms = (time.perf_counter() - t0) * 1000
            resp_body = json.loads(res.read())
            status = res.status

        if status == 202:
            print(f"\n  결과  : {GRN}{BLD}✓ 성공 (202 Accepted){RST}")
        else:
            print(f"\n  결과  : {YEL}HTTP {status} (예상은 202){RST}")

        print(f"  RTT   : {BLD}{rtt_ms:.1f} ms{RST}")
        print(f"  응답  : {resp_body}")
        return status == 202

    except urllib.error.HTTPError as e:
        body = e.read().decode(errors="replace")
        print(f"\n  결과  : {RED}{BLD}✗ HTTP {e.code}{RST}  — {body}")
        return False
    except urllib.error.URLError as e:
        print(f"\n  결과  : {RED}{BLD}✗ 연결 실패{RST}  — {e.reason}")
        return False


def main():
    parser = argparse.ArgumentParser(
        description="스마트팩토리 TCP/IP 로컬 통신 검증 (통신규약 v0.1.0)"
    )
    # 기존 인자명(--server)은 절대 바꾸지 않음
    parser.add_argument(
        "--server",
        default=DEFAULT_SERVER,
        help=f"메인 PC 서버 주소 (기본값: {DEFAULT_SERVER}, 환경변수 MOBOT_SERVER_URL)"
    )
    parser.add_argument(
        "--device",
        default=DEFAULT_DEVICE_ID,
        help=f"장비 식별자 (기본값: {DEFAULT_DEVICE_ID}, 환경변수 MOBOT_DEVICE_ID, 허용: LINE/ARM/AGV)"
    )
    args = parser.parse_args()
    base = args.server.rstrip("/")
    device_id = args.device
    local_ip = get_local_ip()

    print(f"\n{BLD}{'═'*52}{RST}")
    print(f"{BLD}  스마트팩토리 TCP/IP 통신 검증 (통신규약 v0.1.0){RST}")
    print(f"{'═'*52}")
    print(f"  서버      : {BLD}{base}{RST}")
    print(f"  장비 ID   : {BLD}{device_id}{RST}")
    print(f"  로컬 IP   : {BLD}{local_ip}{RST}  (이 기기)")
    print(f"{'═'*52}")

    r1 = test_ping(base)
    time.sleep(0.3)
    r2 = test_device_event(base, device_id)

    print(f"\n{BLD}{'━'*52}{RST}")
    print(f"{BLD}  최종 결과{RST}")
    print(f"{'━'*52}")
    print(f"  Ping 헬스체크          : {GRN+'✓ 성공'+RST if r1 else RED+'✗ 실패'+RST}")
    print(f"  v0.1.0 이벤트 전송     : {GRN+'✓ 성공'+RST if r2 else RED+'✗ 실패'+RST}")
    print(f"{'━'*52}")

    if r1 and r2:
        print(f"\n  {GRN}{BLD}✓ 컴퓨터 ↔ 라즈베리파이 TCP/IP 통신 완전 정상!{RST}")
        print(f"  통신규약 v0.1.0 로컬 내부망 환경 준비 완료.\n")
        sys.exit(0)
    else:
        print(f"\n  {RED}{BLD}✗ 일부 항목 실패. 위 [확인 사항]을 점검하세요.{RST}\n")
        sys.exit(1)


if __name__ == "__main__":
    main()
